        1. sudo pacman -S brightnessctl
        2. in hyprland.lua
        a. hl.bind("XF86MonBrightnessUp",  hl.dsp.exec_cmd("brightnessctl -e4 -n2 set 5%+"),                  { locked = true, repeating = true })
        b. hl.bind("XF86MonBrightnessDown",hl.dsp.exec_cmd("brightnessctl -e4 -n2 set 5%-"),                  { locked = true, repeating = true })

##Cursor:
A. 1. yay -S bibata-cursor-theme-bin
   2. hyprland.lua: a.
        hl.env("XCURSOR_THEME", "Bibata-Modern-Classic")
        hl.env("XCURSOR_SIZE", "24")
        hl.env("HYPRCURSOR_THEME", "Bibata-Modern-Classic")
        hl.env("HYPRCURSOR_SIZE", "24")
                    b.
        hl.on("hyprland.start", function ()
                hl.exec_cmd("hyprctl setcursor Bibata-Modern-Classic 24")
        end)
B. Fix GTK apps:
   1. sudo pacman -S nwg-look
   2. nwg-look
   3. Go to the Mouse Cursor tab, select your installed theme, match the size, and click Apply.
   (alt terminal shortcut) gsettings set org.gnome.desktop.interface cursor-theme 'Bibata-Modern-Classic'
